cmake_minimum_required(VERSION 3.21)
project(Solitare VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(SOLITARE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SOLITARE_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets)
add_library(solitare_core src/core/game.cpp src/core/game_rules.cpp src/core/game_persistence.cpp src/core/theme.cpp src/core/theme_source.cpp src/core/storage.cpp)
target_include_directories(solitare_core PUBLIC src)
target_link_libraries(solitare_core PUBLIC Qt6::Core Qt6::Gui)
add_library(solitare_ui src/ui/card_art.cpp src/ui/floral_art.cpp src/ui/board.cpp src/ui/board_render.cpp src/ui/board_art_cache.cpp src/ui/board_layout.cpp src/ui/modal.cpp src/ui/modal_backdrop.cpp src/ui/modal_motion.cpp src/ui/themed_combo.cpp src/ui/window.cpp src/ui/theme_watcher.cpp)
target_link_libraries(solitare_ui PUBLIC solitare_core Qt6::Widgets Qt6::OpenGLWidgets)
add_executable(solitare src/main.cpp src/cli.cpp)
target_link_libraries(solitare PRIVATE solitare_ui)
foreach(target solitare_core solitare_ui solitare)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    foreach(name game mode theme ui)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        if(name STREQUAL "game" OR name STREQUAL "mode")
            target_link_libraries(${name}_tests PRIVATE solitare_core Qt6::Test)
        else()
            target_link_libraries(${name}_tests PRIVATE solitare_ui Qt6::Test)
        endif()
        add_test(NAME ${name} COMMAND ${name}_tests)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
    add_test(NAME install COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh ${CMAKE_CURRENT_SOURCE_DIR} $<TARGET_FILE_DIR:solitare>)
    add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:solitare>)
endif()
include(GNUInstallDirs)
install(TARGETS solitare RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.solitare.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.solitare.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES assets/PROVENANCE.md assets/UPSTREAM-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DOCDIR})
